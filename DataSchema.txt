CREATE TABLE IF NOT EXISTS stock_daily_data (
    id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '主键ID',
    stock_code VARCHAR(20) NOT NULL COMMENT '股票代码',
    trade_date DATE NOT NULL COMMENT '交易日期',
    open DECIMAL(10, 2) NOT NULL COMMENT '开盘价',
    close DECIMAL(10, 2) NOT NULL COMMENT '收盘价',
    high DECIMAL(10, 2) NOT NULL COMMENT '最高价',
    low DECIMAL(10, 2) NOT NULL COMMENT '最低价',
    volume BIGINT NOT NULL COMMENT '成交量',
    boll_upper DECIMAL(10, 2) COMMENT '布林线上轨',
    boll_middle DECIMAL(10, 2) COMMENT '布林线中轨',
    boll_lower DECIMAL(10, 2) COMMENT '布林线下轨',
    boll_touch_type VARCHAR(10) COMMENT '触碰类型: UPPER(上轨)/LOWER(下轨)/NULL(未触碰)',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '记录更新时间',
    
    -- 联合唯一索引：同一股票同一天只能有一条记录
    UNIQUE KEY uk_stock_date (stock_code, trade_date),
    
    -- 查询优化索引
    INDEX idx_stock_code (stock_code),
    INDEX idx_trade_date (trade_date),
    INDEX idx_boll_touch (boll_touch_type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='股票日线数据表';
